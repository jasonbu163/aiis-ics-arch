# 独立参考client仅链接冻结上游；不进入产品链接，也不拥有服务端image。
cmake_minimum_required(VERSION 4.0)
project(ds_s7_reference LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT APPLE OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "arm64|aarch64")
 message(FATAL_ERROR "Reference target validated only on macOS arm64")
endif()
if(NOT DS_NATIVE_OUTPUT_DIR)
 message(FATAL_ERROR "Explicit DS_NATIVE_OUTPUT_DIR required")
endif()
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${DS_NATIVE_OUTPUT_DIR})
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY ${DS_NATIVE_OUTPUT_DIR})
set(CMAKE_BUILD_RPATH "@loader_path")
set(CMAKE_INSTALL_RPATH "@loader_path")
set(CMAKE_BUILD_WITH_INSTALL_RPATH TRUE)
add_compile_options(-O2 -g -fno-omit-frame-pointer)
set(root ${CMAKE_CURRENT_SOURCE_DIR}/../../native/vendor/snap7-1.4.2/upstream)
set(files)
foreach(file sys/snap_msgsock.cpp sys/snap_sysutils.cpp sys/snap_tcpsrvr.cpp sys/snap_threads.cpp core/s7_client.cpp core/s7_isotcp.cpp core/s7_micro_client.cpp core/s7_partner.cpp core/s7_peer.cpp core/s7_server.cpp core/s7_text.cpp lib/snap7_libmain.cpp)
 list(APPEND files ${root}/src/${file})
endforeach()
add_library(ds_s7_reference SHARED ${files})
target_include_directories(ds_s7_reference PUBLIC ${root}/src/sys ${root}/src/core ${root}/src/lib ${root}/release/Wrappers/c-cpp)
add_executable(ds-s7-reference reference_main.cpp wire_cases.cpp resource_sample.cpp)
target_link_libraries(ds-s7-reference PRIVATE ds_s7_reference)
