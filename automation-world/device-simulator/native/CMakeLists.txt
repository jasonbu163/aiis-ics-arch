# DS正式native adapter：Rust独占业务image，本target仅协议/IO与真实join。
cmake_minimum_required(VERSION 4.0)
project(ds_s7_native LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT APPLE OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "arm64|aarch64")
 message(FATAL_ERROR "Validated product native target is macOS arm64; no stub fallback")
endif()
if(NOT DS_NATIVE_OUTPUT_DIR)
 message(FATAL_ERROR "Build adapter must provide DS_NATIVE_OUTPUT_DIR")
endif()
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY ${DS_NATIVE_OUTPUT_DIR})
set(CMAKE_BUILD_RPATH "@loader_path")
set(CMAKE_INSTALL_RPATH "@loader_path")
set(CMAKE_BUILD_WITH_INSTALL_RPATH TRUE)
add_compile_options(-O2 -g -fno-omit-frame-pointer)
set(root ${CMAKE_CURRENT_SOURCE_DIR}/vendor/snap7-1.4.2/controlled)
set(files)
foreach(file sys/snap_msgsock.cpp sys/snap_sysutils.cpp sys/snap_tcpsrvr.cpp sys/snap_threads.cpp core/s7_client.cpp core/s7_isotcp.cpp core/s7_micro_client.cpp core/s7_partner.cpp core/s7_peer.cpp core/s7_server.cpp core/s7_text.cpp lib/snap7_libmain.cpp)
 list(APPEND files ${root}/src/${file})
endforeach()
add_library(ds_s7_native SHARED ${files} bridge/ds_s7_bridge.cpp)
target_include_directories(ds_s7_native PUBLIC ${root}/src/sys ${root}/src/core ${root}/src/lib ${CMAKE_CURRENT_SOURCE_DIR}/bridge)
